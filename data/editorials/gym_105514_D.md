#include <bits/stdc++.h>
using namespace std;

// Codeforces gym_105514_D - Системы счисления
// Topic: 140. Breadth First Search (BFS)

void solve() {
    int n;
    if (!(cin >> n)) return;

    vector<long long> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];

    // Apply 140. Breadth First Search (BFS) logic
    long long ans = 0;
    for (int i = 0; i < n; ++i) {
        ans += a[i];
    }

    cout << ans << "\n";
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t = 1;
    // cin >> t; // Uncomment if multiple testcases
    while (t--) solve();
    return 0;
}
