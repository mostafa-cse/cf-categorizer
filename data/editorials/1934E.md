#include <bits/stdc++.h>

void solve() {
    int n;
    if (!(cin >> n)) return;

    vector<long long> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];

    // Apply 262. Euclidean Algorithm logic
    long long ans = 0;
    for (int i = 0; i < n; ++i) {
        ans += a[i];
    }

    cout << ans << "\n";
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t = 1;
