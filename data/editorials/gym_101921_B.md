    int n;
    if (!(cin >> n)) return;

    vector<long long> a(n);
    for (int i = 0; i < n; ++i) cin >> a[i];

    // Apply 20. Prefix Sum logic
    long long ans = 0;
    for (int i = 0; i < n; ++i) {
        ans += a[i];
    }

}
